        gsap.registerPlugin(ScrollTrigger);
        const bg = document.getElementById("bg");
        window.addEventListener("scroll", () => {
            const y = window.scrollY;
            bg.style.transform = `scale(${1 + y / 2500}) translateY(${y / 6}px)`;
            if (y < window.innerHeight * 1) bg.style.background = "var(--gradient1)";
            else if (y < window.innerHeight * 2) bg.style.background = "linear-gradient(120deg, #dff6ff 0%, #f8f8f8 100%)";
            else if (y < window.innerHeight * 3) bg.style.background = "linear-gradient(120deg, #101820 0%, #0b0b0b 100%)";
            else bg.style.background = "linear-gradient(120deg, #000 0%, #0a0a0a 100%)";
        });

        const observer = new IntersectionObserver(entries => {
            entries.forEach(entry => {
                if (entry.isIntersecting) entry.target.classList.add('in-view');
            });
        }, { threshold: 0.2 });

        document.querySelectorAll('.fade-in-up, .about-photo img').forEach(el => observer.observe(el));

        gsap.utils.toArray(".timeline-step, .skill-item, .achievement").forEach(el => {
            gsap.to(el, {
                scrollTrigger: { trigger: el, start: "top 85%" },
                opacity: 1, y: 0, duration: 1.2, ease: "power2.out"
            });
        });
    </script>
